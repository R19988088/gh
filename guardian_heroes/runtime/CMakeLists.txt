# Guardian Heroes host runtime, derived from the retained BSD-licensed sources.
# Runtime classes, native platform services and graphics are maintained here.
project(GuardianHeroesHostRuntime VERSION 0.10.0 LANGUAGES C CXX)
if(REXGLUE_PLATFORM_SWITCH)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "The Switch runtime requires devkitA64 GCC")
    endif()
    set(REXGLUE_ENABLE_TRACY OFF)
    set(REXGLUE_ENABLE_PERF_COUNTERS OFF)
    set(REXGLUE_USE_D3D12 OFF)
    set(REXGLUE_USE_VULKAN OFF)
elseif(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(Apple)?Clang$")
    message(FATAL_ERROR "The retained PPC code and host runtime require Clang")
endif()
if(APPLE)
    enable_language(OBJC OBJCXX)
endif()
include(GNUInstallDirs)
set(REXGLUE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(REXGLUE_ENABLE_TRACY ON)
set(REXGLUE_ENABLE_PERF_COUNTERS ON)
set(REXGLUE_BUILD_TESTS OFF)
set(REXGLUE_ENABLE_FIDELITYFX OFF)
if(WIN32)
    option(REXGLUE_USE_D3D12 "Build the D3D12 device backend" ON)
    option(REXGLUE_USE_VULKAN "Build the Vulkan device backend" OFF)
else()
    set(REXGLUE_USE_D3D12 OFF)
    option(REXGLUE_USE_VULKAN "Build the Vulkan device backend" ON)
endif()
if(NOT REXGLUE_PLATFORM_SWITCH AND NOT REXGLUE_USE_VULKAN AND NOT REXGLUE_USE_D3D12)
    message(FATAL_ERROR "A native graphics backend must be enabled")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-fno-strict-aliasing -fno-char8_t)
if(CMAKE_CXX_COMPILER_ID MATCHES "^(Apple)?Clang$")
    add_compile_options(-ffp-model=strict)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    add_compile_options(-ffp-contract=off)
endif()
if(NOT WIN32)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$" AND NOT APPLE)
        add_compile_options(-mcmodel=large)
    else()
        add_compile_options(-mcmodel=small)
    endif()
endif()
add_compile_definitions(REXGLUE_BUILD_CONFIG="$<CONFIG>"
    $<$<NOT:$<CONFIG:Release>>:REXGLUE_ENABLE_PROFILING>
    $<$<NOT:$<CONFIG:Release>>:REXGLUE_ENABLE_PERF_COUNTERS>)
include(cmake/rex_pch.cmake)
add_subdirectory(thirdparty)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../src)
add_subdirectory(src/core)
add_subdirectory(src/filesystem)
add_subdirectory(src/ui)
add_subdirectory(src/input)
add_subdirectory(src/audio)
add_subdirectory(src/system)
add_subdirectory(src/kernel)
add_subdirectory(src/graphics)
# Export only project-owned targets and runtime feature settings to the app.
set(REXGLUE_USE_VULKAN ${REXGLUE_USE_VULKAN} PARENT_SCOPE)
set(REXGLUE_USE_D3D12 ${REXGLUE_USE_D3D12} PARENT_SCOPE)
set(GUARDIAN_HEROES_RUNTIME_ROOT ${CMAKE_CURRENT_SOURCE_DIR} PARENT_SCOPE)
